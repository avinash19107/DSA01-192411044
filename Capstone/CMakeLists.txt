cmake_minimum_required(VERSION 3.20)
project(FORENSIX_CPP VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Crow CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(forensix src/main.cpp)
target_link_libraries(forensix PRIVATE Crow::Crow OpenSSL::Crypto)
if(WIN32)
  target_compile_definitions(forensix PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
